stage('Create Local Tags') {

    steps {

        sh '''
#!/bin/bash
set -e

create_local_tag() {

    REPO=$1
    BRANCH=$2
    AWS_TAG=$3
    LOCAL_TAG=$4

    echo "======================================"
    echo "Repository : $REPO"
    echo "Branch     : $BRANCH"
    echo "AWS Tag    : $AWS_TAG"
    echo "Local Tag  : $LOCAL_TAG"
    echo "Tag Action : $TAG_ACTION"
    echo "======================================"

    rm -rf temp_repo

    git clone "$REPO" temp_repo

    cd temp_repo

    git fetch --all --tags

    git config user.name "Jenkins"
    git config user.email "jenkins@girmiti.in"

    TAG_EXISTS=false

    if git ls-remote --tags origin | grep -q "refs/tags/${LOCAL_TAG}$"
    then
        TAG_EXISTS=true
    fi

    #######################################################
    # DELETE EXISTING TAG (IF REQUIRED)
    #######################################################

    if [ "$TAG_ACTION" = "DELETE_AND_CREATE" ]
    then

        if [ "$TAG_EXISTS" = true ]
        then

            echo "Deleting Remote Tag : $LOCAL_TAG"

            git push origin --delete "$LOCAL_TAG" || true

            echo "Deleting Local Tag : $LOCAL_TAG"

            git tag -d "$LOCAL_TAG" || true

            git fetch --prune --tags

        else

            echo "Tag does not exist. Creating new tag."

        fi

else

    if [ "$TAG_EXISTS" = true ]
    then

        echo "======================================"
        echo "Tag already exists."
        echo "Skipping tag creation."
        echo "Checking out existing tag : $LOCAL_TAG"
        echo "======================================"

        git checkout "$LOCAL_TAG"

        cd .. 
        rm -rf temp_repo 

        return

    fi
fi
    #######################################################
    # CHECKOUT SOURCE
    #######################################################

    if [ "$SOURCE_TYPE" = "BRANCH" ]
    then

        echo "Checking out Branch : $BRANCH"

        git checkout "$BRANCH"

    else

        echo "Checking out AWS Tag : $AWS_TAG"

        git checkout "$AWS_TAG"

    fi

    #######################################################
    # CREATE LOCAL TAG
    #######################################################

    echo "Creating Local Tag : $LOCAL_TAG"

    git tag -a "$LOCAL_TAG" -m "Created from Jenkins"

    git push origin "$LOCAL_TAG"

    cd ..

    rm -rf temp_repo

    echo "======================================"

}

########################################################
# SERVICE REPOSITORY
########################################################

create_local_tag \
"$SERVICE_REPO" \
"$SERVICE_BRANCH" \
"$SERVICE_TAG" \
"$LOCAL_SERVICE_TAG"

########################################################
# DOCUMENT REPOSITORY
########################################################

create_local_tag \
"$DOCUMENT_REPO" \
"$DB_BRANCH" \
"$DB_TAG" \
"$LOCAL_DB_TAG"

echo
echo "======================================"
echo "Local Tag Creation Completed"
echo "======================================"

'''
    }

done

echo
